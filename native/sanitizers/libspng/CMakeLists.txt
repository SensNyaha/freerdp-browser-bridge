cmake_minimum_required(VERSION 3.24)
project(SpngRegression LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Standalone local test project. The root/Release build does not include it.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This pinned fmemopen corpus profile is Linux-only")
endif()
set(BRIDGE_SPNG_SOURCE "" CACHE PATH "Prepared pinned libspng 0.7.4 source")
set(harness "${BRIDGE_SPNG_SOURCE}/tests/spng_read_fuzzer.c")
if(NOT EXISTS "${harness}" OR IS_DIRECTORY "${harness}" OR IS_SYMLINK "${harness}")
  message(FATAL_ERROR "Expected the regular pinned libspng read harness")
endif()
file(SIZE "${harness}" harness_size)
if(harness_size GREATER 16384)
  message(FATAL_ERROR "Pinned read harness exceeds its bound")
endif()
file(SHA256 "${harness}" harness_hash)
if(NOT harness_hash STREQUAL "3ed658e5b3dab357d1e9078d3cdd930fe0e4e4965755bd2ee2b06b5074a4d122")
  message(FATAL_ERROR "Pinned libspng read harness hash mismatch")
endif()
find_package(SPNG 0.7.4 EXACT CONFIG REQUIRED)
add_library(spng-read-harness OBJECT "${harness}")
target_compile_definitions(spng-read-harness PRIVATE _GNU_SOURCE SPNGT_HAVE_FMEMOPEN)
add_executable(spng-regression regression.cpp $<TARGET_OBJECTS:spng-read-harness>)
target_link_libraries(spng-regression PRIVATE spng::spng)
target_compile_options(spng-regression PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(BRIDGE_SPNG_SANITIZERS "Instrument the independent PNG regression" ON)
if(BRIDGE_SPNG_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang)$")
    message(FATAL_ERROR "The sanitizer profile requires GCC or Clang")
  endif()
  foreach(target spng-read-harness spng-regression)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined
      -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(spng-regression PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME spng-roundtrip COMMAND spng-regression)
add_test(NAME spng-upstream-corpus COMMAND spng-regression
  "${BRIDGE_SPNG_SOURCE}/tests/images" "${BRIDGE_SPNG_SOURCE}/tests/crashers")
set_tests_properties(spng-roundtrip spng-upstream-corpus PROPERTIES TIMEOUT 90)
set_tests_properties(spng-roundtrip spng-upstream-corpus PROPERTIES ENVIRONMENT
  "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:spng::spng>:$<TARGET_FILE_DIR:ZLIB::ZLIB>")
