cmake_minimum_required(VERSION 3.24)
project(FreeRDPNodeBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BRIDGE_SANITIZERS "Enable ASan and UBSan" OFF)
option(BRIDGE_FUZZ "Build libFuzzer IPC target (Clang)" OFF)
option(BRIDGE_STATIC_ANALYSIS "Enable MSVC static analysis for bridge-owned code" ON)
option(BRIDGE_WITH_FREERDP "Link exact pinned FreeRDP libraries and experimental connection core" OFF)
set(BRIDGE_DEPENDENCY_PREFIX "" CACHE PATH "Private native dependency runtime prefix")
set(BRIDGE_FREERDP_PREFIX "" CACHE PATH "Private pinned FreeRDP/WinPR runtime prefix")
include(CTest)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
  OUTPUT_VARIABLE BRIDGE_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT BRIDGE_GIT_COMMIT)
  set(BRIDGE_GIT_COMMIT "uncommitted")
else()
  execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE BRIDGE_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(BRIDGE_DIRTY)
    string(APPEND BRIDGE_GIT_COMMIT "-dirty")
  endif()
endif()
add_library(bridge_warnings INTERFACE)
if(MSVC)
  target_compile_options(bridge_warnings INTERFACE /W4 /WX /permissive- /utf-8)
  target_compile_options(bridge_warnings INTERFACE /external:anglebrackets /external:W0)
  if(BRIDGE_STATIC_ANALYSIS)
    target_compile_options(bridge_warnings INTERFACE /analyze /analyze:external-)
  endif()
else()
  target_compile_options(bridge_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
  if(BRIDGE_SANITIZERS)
    target_compile_options(bridge_warnings INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bridge_warnings INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(bridge_ipc native/ipc/protocol.cpp native/ipc/connection_config.cpp)
target_include_directories(bridge_ipc PUBLIC native/ipc)
target_link_libraries(bridge_ipc PRIVATE bridge_warnings)
add_library(bridge_avc native/media/avc.cpp)
target_include_directories(bridge_avc PUBLIC native/media)
target_link_libraries(bridge_avc PUBLIC bridge_ipc PRIVATE bridge_warnings)
add_executable(rdp-worker native/rdp-worker/main.cpp)
target_link_libraries(rdp-worker PRIVATE bridge_ipc bridge_warnings)
target_compile_definitions(rdp-worker PRIVATE BRIDGE_VERSION="${PROJECT_VERSION}" BRIDGE_GIT_COMMIT="${BRIDGE_GIT_COMMIT}")
set(BRIDGE_BUILD_FEATURES "{\"enabledChannels\":[],\"codecProviders\":[],\"dependencies\":{}}")
if(BRIDGE_WITH_FREERDP)
  find_package(FreeRDP 3.32.1 EXACT CONFIG REQUIRED)
  find_package(FreeRDP-Client 3.32.1 EXACT CONFIG REQUIRED)
  find_package(WinPR 3.32.1 EXACT CONFIG REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(libjpeg-turbo CONFIG REQUIRED)
  add_library(bridge_jpeg native/media/jpeg.cpp)
  target_include_directories(bridge_jpeg PUBLIC native/media)
  target_link_libraries(bridge_jpeg PUBLIC bridge_ipc libjpeg-turbo::turbojpeg PRIVATE bridge_warnings)
  find_path(BRIDGE_OPENH264_INCLUDE wels/codec_api.h PATHS "${BRIDGE_DEPENDENCY_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(BRIDGE_OPENH264_LIBRARY NAMES openh264 PATHS "${BRIDGE_DEPENDENCY_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(bridge_openh264_runtime INTERFACE)
  target_include_directories(bridge_openh264_runtime SYSTEM INTERFACE "${BRIDGE_OPENH264_INCLUDE}")
  target_link_libraries(bridge_openh264_runtime INTERFACE "${BRIDGE_OPENH264_LIBRARY}")
  add_library(bridge_h264 native/media/openh264.cpp)
  target_link_libraries(bridge_h264 PUBLIC bridge_avc bridge_openh264_runtime PRIVATE bridge_warnings)
  target_sources(rdp-worker PRIVATE native/rdp-worker/connection.cpp native/rdp-worker/certificate.cpp)
  target_link_libraries(rdp-worker PRIVATE freerdp freerdp-client winpr OpenSSL::Crypto bridge_jpeg)
  if(WIN32)
    target_link_libraries(rdp-worker PRIVATE ws2_32)
  endif()
  target_compile_definitions(rdp-worker PRIVATE BRIDGE_WITH_FREERDP=1)
  find_package(ZLIB REQUIRED)
  add_executable(rdp-runtime-check native/runtime/dependency_probe.cpp)
  target_link_libraries(rdp-runtime-check PRIVATE ZLIB::ZLIB freerdp freerdp-client bridge_jpeg bridge_h264 bridge_warnings ${CMAKE_DL_LIBS})
  set_target_properties(rdp-worker rdp-runtime-check PROPERTIES INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN")
  if(WIN32)
    file(GLOB BRIDGE_RUNTIME_DLLS "${BRIDGE_DEPENDENCY_PREFIX}/bin/*.dll" "${BRIDGE_FREERDP_PREFIX}/bin/*.dll")
    foreach(target rdp-worker rdp-runtime-check)
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${BRIDGE_RUNTIME_DLLS} "$<TARGET_FILE_DIR:${target}>")
    endforeach()
  endif()
  install(TARGETS rdp-runtime-check RUNTIME DESTINATION bin)
  if(NOT EXISTS "${CMAKE_SOURCE_DIR}/build/native-features.json")
    message(FATAL_ERROR "Generate the actual dependency feature report before the bundled build")
  endif()
  file(READ "${CMAKE_SOURCE_DIR}/build/native-features.json" BRIDGE_BUILD_FEATURES)
endif()
configure_file(native/runtime/build_features.hpp.in generated/build_features.hpp @ONLY)
target_include_directories(rdp-worker PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.8.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(Catch2)
  add_executable(ipc-tests native/ipc/tests.cpp)
  target_link_libraries(ipc-tests PRIVATE bridge_ipc bridge_warnings Catch2::Catch2WithMain)
  add_test(NAME ipc-tests COMMAND ipc-tests)
  add_executable(avc-tests native/media/avc_tests.cpp)
  target_link_libraries(avc-tests PRIVATE bridge_avc bridge_warnings Catch2::Catch2WithMain)
  add_test(NAME avc-tests COMMAND avc-tests)
  if(BRIDGE_WITH_FREERDP)
    add_executable(jpeg-tests native/media/tests.cpp)
    target_link_libraries(jpeg-tests PRIVATE bridge_jpeg bridge_warnings Catch2::Catch2WithMain)
    add_test(NAME jpeg-tests COMMAND jpeg-tests)
    add_executable(h264-tests native/media/openh264_tests.cpp)
    target_include_directories(h264-tests PRIVATE native/runtime)
    target_link_libraries(h264-tests PRIVATE bridge_h264 bridge_warnings Catch2::Catch2WithMain)
    add_test(NAME h264-tests COMMAND h264-tests)
  endif()
  foreach(mode RANGE 1 3)
    add_executable(fixture-worker-${mode} tests/fixtures/unresponsive.cpp)
    target_link_libraries(fixture-worker-${mode} PRIVATE bridge_ipc bridge_warnings)
    target_compile_definitions(fixture-worker-${mode} PRIVATE FIXTURE_MODE=${mode})
  endforeach()
endif()
if(BRIDGE_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "BRIDGE_FUZZ requires Clang")
  endif()
  add_executable(ipc-fuzz native/ipc/fuzz.cpp native/ipc/protocol.cpp native/ipc/connection_config.cpp)
  target_include_directories(ipc-fuzz PRIVATE native/ipc)
  target_compile_options(ipc-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(ipc-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS rdp-worker RUNTIME DESTINATION bin)
include(native/packaging.cmake)
