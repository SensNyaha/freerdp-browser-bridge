cmake_minimum_required(VERSION 3.24)
project(FreeRDPNodeBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BRIDGE_SANITIZERS "Enable ASan and UBSan" OFF)
option(BRIDGE_FUZZ "Build libFuzzer IPC target (Clang)" OFF)
option(BRIDGE_STATIC_ANALYSIS "Enable MSVC static analysis for bridge-owned code" ON)
option(BRIDGE_WITH_FREERDP "Link exact pinned FreeRDP libraries (connection not implemented yet)" OFF)
include(CTest)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
  OUTPUT_VARIABLE BRIDGE_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT BRIDGE_GIT_COMMIT)
  set(BRIDGE_GIT_COMMIT "uncommitted")
else()
  execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE BRIDGE_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(BRIDGE_DIRTY)
    string(APPEND BRIDGE_GIT_COMMIT "-dirty")
  endif()
endif()
add_library(bridge_warnings INTERFACE)
if(MSVC)
  target_compile_options(bridge_warnings INTERFACE /W4 /WX /permissive- /utf-8)
  if(BRIDGE_STATIC_ANALYSIS)
    target_compile_options(bridge_warnings INTERFACE /analyze)
  endif()
else()
  target_compile_options(bridge_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
  if(BRIDGE_SANITIZERS)
    target_compile_options(bridge_warnings INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bridge_warnings INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(bridge_ipc native/ipc/protocol.cpp)
target_include_directories(bridge_ipc PUBLIC native/ipc)
target_link_libraries(bridge_ipc PRIVATE bridge_warnings)
add_executable(rdp-worker native/rdp-worker/main.cpp)
target_link_libraries(rdp-worker PRIVATE bridge_ipc bridge_warnings)
target_compile_definitions(rdp-worker PRIVATE BRIDGE_VERSION="${PROJECT_VERSION}" BRIDGE_GIT_COMMIT="${BRIDGE_GIT_COMMIT}")
if(BRIDGE_WITH_FREERDP)
  find_package(FreeRDP 3.32.1 EXACT CONFIG REQUIRED)
  find_package(FreeRDP-Client 3.32.1 EXACT CONFIG REQUIRED)
  find_package(WinPR 3.32.1 EXACT CONFIG REQUIRED)
  target_link_libraries(rdp-worker PRIVATE freerdp freerdp-client winpr)
  target_compile_definitions(rdp-worker PRIVATE BRIDGE_WITH_FREERDP=1)
endif()
if(BUILD_TESTING)
  include(FetchContent)
  FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.8.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(Catch2)
  add_executable(ipc-tests native/ipc/tests.cpp)
  target_link_libraries(ipc-tests PRIVATE bridge_ipc bridge_warnings Catch2::Catch2WithMain)
  add_test(NAME ipc-tests COMMAND ipc-tests)
  foreach(mode RANGE 1 3)
    add_executable(fixture-worker-${mode} tests/fixtures/unresponsive.cpp)
    target_link_libraries(fixture-worker-${mode} PRIVATE bridge_ipc bridge_warnings)
    target_compile_definitions(fixture-worker-${mode} PRIVATE FIXTURE_MODE=${mode})
  endforeach()
endif()
if(BRIDGE_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "BRIDGE_FUZZ requires Clang")
  endif()
  add_executable(ipc-fuzz native/ipc/fuzz.cpp native/ipc/protocol.cpp)
  target_include_directories(ipc-fuzz PRIVATE native/ipc)
  target_compile_options(ipc-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(ipc-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS rdp-worker RUNTIME DESTINATION bin)
